"""
Script para generar el documento PDF de requerimientos del sistema POS SERVICE.
Ejecutar: python generar-requerimientos.txt
Salida:   POS-Requerimientos-Sistema.pdf (en la misma carpeta docs/)
Requisito: pip install fpdf2
"""


from pathlib import Path

from fpdf import FPDF

OUTPUT_NAME = "POS-Requerimientos-Sistema.pdf"


class PosPdf(FPDF):
    def __init__(self):
        super().__init__()
        self.doc_title = "POS SERVICE - Documento de Requerimientos"
        self.set_auto_page_break(auto=True, margin=20)

    def header(self):
        if self.page_no() == 1:
            return
        self.set_font("Helvetica", "I", 8)
        self.set_text_color(100, 100, 100)
        self.cell(0, 8, self.doc_title, align="C", new_x="LMARGIN", new_y="NEXT")
        self.line(10, self.get_y(), 200, self.get_y())
        self.ln(4)

    def footer(self):
        self.set_y(-15)
        self.set_font("Helvetica", "I", 8)
        self.set_text_color(120, 120, 120)
        self.cell(0, 10, f"Pagina {self.page_no()}/{{nb}}", align="C")

    def cover_page(self, title, subtitle, version):
        self.add_page()
        self.ln(50)
        self.set_font("Helvetica", "B", 24)
        self.set_text_color(30, 60, 120)
        self.multi_cell(0, 12, title, align="C")
        self.ln(8)
        self.set_font("Helvetica", "", 14)
        self.set_text_color(60, 60, 60)
        self.multi_cell(0, 8, subtitle, align="C")
        self.ln(20)
        self.set_font("Helvetica", "I", 11)
        self.cell(0, 8, version, align="C", new_x="LMARGIN", new_y="NEXT")
        self.ln(30)
        self.set_font("Helvetica", "", 10)
        self.set_text_color(80, 80, 80)
        self.multi_cell(
            0,
            6,
            "Documento generado automaticamente a partir del analisis del repositorio "
            "POS SERVICE (backend Spring Boot + frontend Angular 20).",
            align="C",
        )

    def chapter_title(self, number, title):
        self.add_page()
        self.set_font("Helvetica", "B", 16)
        self.set_text_color(30, 60, 120)
        self.cell(0, 10, f"{number}. {title}", new_x="LMARGIN", new_y="NEXT")
        self.ln(2)
        self.set_draw_color(30, 60, 120)
        self.line(10, self.get_y(), 200, self.get_y())
        self.ln(6)

    def section_title(self, title):
        self.ln(4)
        self.set_font("Helvetica", "B", 12)
        self.set_text_color(40, 40, 40)
        self.cell(0, 8, title, new_x="LMARGIN", new_y="NEXT")
        self.ln(2)

    def body(self, text):
        self.set_font("Helvetica", "", 10)
        self.set_text_color(30, 30, 30)
        self.multi_cell(0, 6, text)
        self.ln(2)

    def bullet_list(self, items):
        self.set_font("Helvetica", "", 10)
        self.set_text_color(30, 30, 30)
        for item in items:
            self.multi_cell(0, 6, f"  - {item}")
        self.ln(2)

def build_pdf():
    pdf = PosPdf()
    pdf.alias_nb_pages()

    pdf.cover_page(
        "POS SERVICE",
        "Documento de Requerimientos del Sistema",
        "Version 1.0 - Julio 2026",
    )

    # Indice
    pdf.add_page()
    pdf.set_font("Helvetica", "B", 14)
    pdf.set_text_color(30, 60, 120)
    pdf.cell(0, 10, "Indice", new_x="LMARGIN", new_y="NEXT")
    pdf.ln(4)
    pdf.set_font("Helvetica", "", 10)
    pdf.set_text_color(30, 30, 30)
    for i, (title, _) in enumerate(SECTIONS, 1):
        pdf.cell(0, 7, f"  {i}. {title}", new_x="LMARGIN", new_y="NEXT")

    for i, (title, subsections) in enumerate(SECTIONS, 1):
        pdf.chapter_title(str(i), title)
        for sub_title, sub_body in subsections:
            pdf.section_title(sub_title)
            pdf.body(sub_body)

    output_path = Path(__file__).parent / OUTPUT_NAME
    pdf.output(str(output_path))
    print(f"PDF generado: {output_path}")


if __name__ == "__main__":
    build_pdf()
